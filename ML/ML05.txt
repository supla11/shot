# 1. Import Libraries
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# 2. Load Dataset
df = pd.read_csv(r"C:\Users\Satish\Downloads\sales_data_sample-selected-columns.csv")

print(df.head())

# 3. Select Numerical Columns
X = df[['QUANTITYORDERED', 'PRICEEACH', 'SALES']]

# 4. Handle Missing Values
X = X.dropna()

# 5. Feature Scaling
sc = StandardScaler()
X_scaled = sc.fit_transform(X)

# 6. Elbow Method
wcss = []

for i in range(1, 11):
    model = KMeans(n_clusters=i, random_state=42, n_init=10)
    model.fit(X_scaled)
    wcss.append(model.inertia_)

plt.plot(range(1, 11), wcss, marker='o')
plt.title("Elbow Method")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.show()

# 7. Apply K-Means Clustering
kmeans = KMeans(n_clusters=3, random_state=42, n_init=10)
y_pred = kmeans.fit_predict(X_scaled)

# 8. Add Cluster Column
df = df.loc[X.index].copy()
df['Cluster'] = y_pred

# 9. Display Results
print("\nK-Means Clustering Results:")
print(df[['QUANTITYORDERED', 'PRICEEACH', 'SALES', 'Cluster']].head(10))

# 10. Visualize Clusters
plt.scatter(X_scaled[:, 0], X_scaled[:, 2], c=y_pred, cmap='rainbow')
plt.title("K-Means Clustering")
plt.xlabel("Quantity Ordered")
plt.ylabel("Sales")
plt.show()

# 11. Cluster Count
print("\nNumber of Records in Each Cluster:")
print(df['Cluster'].value_counts())

print("\nK-Means Clustering Completed Successfully!")
